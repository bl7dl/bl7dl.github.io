        xalign 0.5
        xanchor 0.5
        yanchor 0.0
    transform left:
        xalign 0.28
        xanchor 0.5
        yanchor 0.0
    transform right:
        xalign 0.72
        xanchor 0.5
        yanchor 0.0
    transform fleft:
        xalign 0.16
        xanchor 0.5
        yanchor 0.0
